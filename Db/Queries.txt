 
Create users table
 CREATE TABLE IF NOT EXISTS users (
      id INT AUTO_INCREMENT PRIMARY KEY,
      name VARCHAR(100) NOT NULL,
      email VARCHAR(150) NOT NULL UNIQUE,
      passwordHash VARCHAR(100) NOT NULL,
      createdOn DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
    ) ENGINE=InnoDB

Create projects table
    CREATE TABLE IF NOT EXISTS projects (
      id INT AUTO_INCREMENT PRIMARY KEY,
      name VARCHAR(150) NOT NULL,
      projectKey VARCHAR(10) NOT NULL UNIQUE,
      ownerId INT NOT NULL,
      createdOn DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (ownerId) REFERENCES users(id)
    ) ENGINE=InnoDB


 Create issues table
      CREATE TABLE IF NOT EXISTS issues (
      id INT AUTO_INCREMENT PRIMARY KEY,
      projectId INT NOT NULL,
      title VARCHAR(255) NOT NULL,
      description TEXT NOT NULL,
      status ENUM('todo','in_progress','in_review','done')
        NOT NULL DEFAULT 'todo',
      priority ENUM('low','medium','high')
        NOT NULL DEFAULT 'medium',
      type ENUM('task','bug','story')
        NOT NULL DEFAULT 'task',
      assigneeId INT NULL,
      reporterId INT NOT NULL,
      createdOn DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
      updatedOn DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
      INDEX idx_project_status (projectId, status),
      INDEX idx_createdOn (createdOn),
      FOREIGN KEY (projectId) REFERENCES projects(id) ON DELETE CASCADE,
      FOREIGN KEY (assigneeId) REFERENCES users(id) ON DELETE SET NULL,
      FOREIGN KEY (reporterId) REFERENCES users(id)
    ) ENGINE=InnoDB

CREATE TABLE IF NOT EXISTS comments (
  id        INT AUTO_INCREMENT PRIMARY KEY,
  issueId   INT NOT NULL,
  userId    INT NOT NULL,
  body      TEXT NOT NULL,
  createdAt DATETIME NOT NULL,
  INDEX idx_comments_issue (issueId)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS activity (
  id        INT AUTO_INCREMENT PRIMARY KEY,
  issueId   INT NOT NULL,
  userId    INT NOT NULL,
  field     VARCHAR(50) NOT NULL,    -- 'created', 'status', 'assigneeId', ...
  oldValue  VARCHAR(255) NULL,
  newValue  VARCHAR(255) NULL,
  createdAt DATETIME NOT NULL,
  INDEX idx_activity_issue (issueId)
) ENGINE=InnoDB;
